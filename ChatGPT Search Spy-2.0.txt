// ==UserScript==
// @name         ChatGPT Search Spy
// @namespace    getkellan
// @version      2.0
// @description  Shows the web search queries + sources ChatGPT uses for an answer
// @match        https://chatgpt.com/*
// @match        https://chat.openai.com/*
// @run-at       document-start
// @grant        none
// ==/UserScript==

(function () {
  'use strict';
  const W = window;
  const queries = new Set();
  const sources = new Map(); // url -> {title, domain}
  let panel, qList, sList, countEl;

  // ---------- extraction ----------
  const QUERY_KEYS = /^(search_queries|search_model_queries|queries|search_query|query)$/i;

  function addQuery(q) {
    if (typeof q !== 'string') return;
    q = q.trim();
    if (q.length < 2 || q.length > 300 || /^https?:/.test(q)) return;
    if (!queries.has(q)) { queries.add(q); render(); }
  }

  function addSource(url, title, domain) {
    if (!url || sources.has(url)) return;
    try { domain = domain || new URL(url).hostname.replace(/^www\./, ''); } catch (e) {}
    sources.set(url, { title: title || url, domain: domain || '' });
    render();
  }

  function scan(node, parentKey) {
    if (!node || typeof node !== 'object') {
      if (typeof node === 'string') scanText(node);
      return;
    }
    if (Array.isArray(node)) { node.forEach(n => scan(n, parentKey)); return; }

    // query objects like {q: "..."} or {query: "..."}
    if (QUERY_KEYS.test(parentKey || '')) {
      if (typeof node.q === 'string') addQuery(node.q);
      if (typeof node.query === 'string') addQuery(node.query);
    }
    // search_result_groups -> entries -> {url,title,attribution}
    if (typeof node.url === 'string' && /^https?:/.test(node.url) &&
        (node.title || node.attribution || node.snippet)) {
      addSource(node.url, node.title, node.attribution);
    }
    for (const [k, v] of Object.entries(node)) {
      if (QUERY_KEYS.test(k)) {
        if (typeof v === 'string') addQuery(v);
        else if (Array.isArray(v)) v.forEach(x => typeof x === 'string' ? addQuery(x) : scan(x, k));
        else scan(v, k);
      } else {
        scan(v, k);
      }
    }
  }

  // tool calls sometimes arrive as raw text: {"search_query":[{"q":"..."}]} or search("...")
  function scanText(t) {
    if (t.length < 4 || !/q"|query|search\(/.test(t)) return;
    let m;
    const r1 = /"(?:q|query)"\s*:\s*"((?:[^"\\]|\\.){2,300})"/g;
    while ((m = r1.exec(t))) { try { addQuery(JSON.parse('"' + m[1] + '"')); } catch (e) { addQuery(m[1]); } }
    const r2 = /search\(\s*["']([^"']{2,300})["']/g;
    while ((m = r2.exec(t))) addQuery(m[1]);
  }

  // ---------- fetch hook ----------
  const origFetch = W.fetch;
  W.fetch = async function (...args) {
    const res = await origFetch.apply(this, args);
    try {
      const url = typeof args[0] === 'string' ? args[0] : (args[0] && args[0].url) || '';
      if (/\/backend-(api|anon)\/(f\/)?conversation/.test(url)) handle(res.clone());
    } catch (e) {}
    return res;
  };

  async function handle(res) {
    const type = res.headers.get('content-type') || '';
    if (type.includes('event-stream')) {
      const reader = res.body.getReader();
      const dec = new TextDecoder();
      let buf = '', streamText = '';
      for (;;) {
        const { value, done } = await reader.read();
        if (done) break;
        buf += dec.decode(value, { stream: true });
        const lines = buf.split('\n');
        buf = lines.pop();
        for (const line of lines) {
          if (!line.startsWith('data:')) continue;
          const data = line.slice(5).trim();
          if (!data || data === '[DONE]') continue;
          try {
            const j = JSON.parse(data);
            scan(j);
            collectStrings(j, s => (streamText += s));
          } catch (e) {}
        }
      }
      scanText(streamText); // catches tool-call JSON streamed in pieces
    } else if (type.includes('json')) {
      scan(await res.json());
    }
  }

  function collectStrings(n, cb) {
    if (typeof n === 'string') cb(n);
    else if (n && typeof n === 'object') Object.values(n).forEach(v => collectStrings(v, cb));
  }

  // reset when switching chats
  let lastPath = location.pathname;
  setInterval(() => {
    if (location.pathname !== lastPath) {
      lastPath = location.pathname;
      queries.clear(); sources.clear(); render();
    }
  }, 500);

  // ---------- UI (shadow DOM, matches ChatGPT look) ----------
  let root, tab = 'queries', open = false;

  const ICON = {
    search: '<svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round"><circle cx="11" cy="11" r="7"/><path d="m20 20-3.5-3.5"/></svg>',
    close: '<svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round"><path d="M18 6 6 18M6 6l12 12"/></svg>',
    copy: '<svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><rect x="9" y="9" width="12" height="12" rx="2.5"/><path d="M5 15V5a2 2 0 0 1 2-2h10"/></svg>',
    check: '<svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.2" stroke-linecap="round" stroke-linejoin="round"><path d="M20 6 9 17l-5-5"/></svg>',
    ext: '<svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M7 17 17 7M8 7h9v9"/></svg>'
  };

  const CSS = `
    :host { all: initial; }
    * { box-sizing: border-box; }
    .wrap { --bg:#fff; --bg2:#f9f9f9; --hover:#ececec; --fg:#0d0d0d; --fg2:#5d5d5d; --fg3:#8f8f8f; --bd:rgba(0,0,0,.1);
      --shadow:0 4px 24px rgba(0,0,0,.12), 0 0 0 1px rgba(0,0,0,.06);
      font: 14px/1.45 ui-sans-serif,-apple-system,system-ui,"Segoe UI",Helvetica,"Apple Color Emoji",Arial,sans-serif;
      color: var(--fg); -webkit-font-smoothing: antialiased; }
    .wrap.dark { --bg:#2f2f2f; --bg2:#262626; --hover:#424242; --fg:#ececec; --fg2:#b4b4b4; --fg3:#8f8f8f; --bd:rgba(255,255,255,.1);
      --shadow:0 8px 32px rgba(0,0,0,.5), 0 0 0 1px rgba(255,255,255,.08); }
    button { font: inherit; color: inherit; background: none; border: 0; cursor: pointer; }
    .fab { position: fixed; right: 20px; bottom: 112px; height: 36px; padding: 0 12px 0 10px; border-radius: 999px;
      display: flex; align-items: center; gap: 6px; background: var(--bg); color: var(--fg); box-shadow: var(--shadow);
      font-size: 13px; font-weight: 500; transition: transform .15s, opacity .15s; z-index: 2147483000; }
    .fab:hover { background: var(--hover); }
    .fab .n { color: var(--fg2); font-variant-numeric: tabular-nums; }
    .fab.hide { opacity: 0; transform: scale(.9); pointer-events: none; }
    .panel { position: fixed; right: 20px; bottom: 112px; width: 360px; max-height: min(560px, 70vh); display: flex; flex-direction: column;
      background: var(--bg); border-radius: 16px; box-shadow: var(--shadow); overflow: hidden; z-index: 2147483000;
      transform-origin: bottom right; transition: transform .18s cubic-bezier(.2,.9,.3,1), opacity .15s; }
    .panel.hide { opacity: 0; transform: scale(.96) translateY(6px); pointer-events: none; }
    .head { display: flex; align-items: center; justify-content: space-between; padding: 12px 12px 8px 16px; }
    .title { font-weight: 600; font-size: 15px; }
    .icon-btn { width: 32px; height: 32px; border-radius: 8px; display: grid; place-items: center; color: var(--fg2); }
    .icon-btn:hover { background: var(--hover); color: var(--fg); }
    .seg { display: flex; gap: 2px; margin: 0 12px 8px; padding: 3px; background: var(--bg2); border-radius: 10px; border: 1px solid var(--bd); }
    .seg button { flex: 1; height: 28px; border-radius: 7px; font-size: 13px; color: var(--fg2); }
    .seg button.on { background: var(--bg); color: var(--fg); box-shadow: 0 1px 2px rgba(0,0,0,.12); }
    .wrap.dark .seg button.on { background: var(--hover); }
    .seg .c { color: var(--fg3); margin-left: 4px; font-variant-numeric: tabular-nums; }
    .list { overflow-y: auto; padding: 0 6px 6px; scrollbar-width: thin; }
    .row { display: flex; align-items: center; gap: 10px; padding: 8px 10px; border-radius: 10px; text-decoration: none; color: var(--fg); }
    .row:hover { background: var(--hover); }
    .row .main { flex: 1; min-width: 0; }
    .row .t { overflow: hidden; text-overflow: ellipsis; white-space: nowrap; }
    .row .s { font-size: 12px; color: var(--fg3); overflow: hidden; text-overflow: ellipsis; white-space: nowrap; }
    .row .idx { width: 20px; height: 20px; flex: none; border-radius: 6px; display: grid; place-items: center;
      font-size: 11px; color: var(--fg2); background: var(--bg2); border: 1px solid var(--bd); }
    .row img { width: 18px; height: 18px; flex: none; border-radius: 4px; }
    .row .act { opacity: 0; color: var(--fg2); width: 28px; height: 28px; border-radius: 7px; display: grid; place-items: center; flex: none; }
    .row:hover .act { opacity: 1; }
    .row .act:hover { color: var(--fg); background: var(--bg2); }
    .empty { padding: 28px 20px 32px; text-align: center; color: var(--fg2); font-size: 13px; }
    .empty b { display: block; color: var(--fg); font-weight: 500; margin-bottom: 4px; font-size: 14px; }
    .foot { display: flex; justify-content: flex-end; padding: 8px 12px 12px; border-top: 1px solid var(--bd); }
    .btn { height: 32px; padding: 0 12px; border-radius: 999px; border: 1px solid var(--bd); display: flex; align-items: center; gap: 6px; font-size: 13px; }
    .btn:hover { background: var(--hover); }
  `;

  function isDark() {
    const h = document.documentElement;
    if (h.classList.contains('dark')) return true;
    if (h.classList.contains('light')) return false;
    const m = getComputedStyle(document.body).backgroundColor.match(/\d+/g);
    if (m) return (+m[0] * 0.299 + +m[1] * 0.587 + +m[2] * 0.114) < 128;
    return matchMedia('(prefers-color-scheme: dark)').matches;
  }

  function build() {
    if (root || !document.body) return;
    const host = document.createElement('div');
    host.id = 'search-spy-host';
    document.body.appendChild(host);
    root = host.attachShadow({ mode: 'open' });
    root.innerHTML = `<style>${CSS}</style>
      <div class="wrap">
        <button class="fab" title="Searches ChatGPT ran">${ICON.search}<span>Searches</span><span class="n"></span></button>
        <div class="panel hide">
          <div class="head"><div class="title">Searches</div><button class="icon-btn x" title="Close">${ICON.close}</button></div>
          <div class="seg"><button data-t="queries">Queries<span class="c"></span></button><button data-t="sources">Sources<span class="c"></span></button></div>
          <div class="list"></div>
          <div class="foot"><button class="btn copyall">${ICON.copy}<span>Copy all</span></button></div>
        </div>
      </div>`;
    const $ = s => root.querySelector(s);
    const toggle = v => { open = v; $('.panel').classList.toggle('hide', !v); $('.fab').classList.toggle('hide', v); };
    $('.fab').onclick = () => toggle(true);
    $('.x').onclick = () => toggle(false);
    root.querySelectorAll('.seg button').forEach(b => b.onclick = () => { tab = b.dataset.t; render(); });
    document.addEventListener('keydown', e => { if (e.key === 'Escape' && open) toggle(false); });
    $('.copyall').onclick = e => {
      const txt = 'QUERIES\n' + [...queries].join('\n') +
        '\n\nSOURCES\n' + [...sources].map(([u, s]) => `${s.domain} | ${s.title} | ${u}`).join('\n');
      flash(e.currentTarget, txt, 'Copied');
    };
    $('.list').addEventListener('click', e => {
      const c = e.target.closest('.act[data-copy]');
      if (!c) return;
      e.preventDefault(); e.stopPropagation();
      flash(c, c.dataset.copy);
    });
    new MutationObserver(applyTheme).observe(document.documentElement, { attributes: true, attributeFilter: ['class', 'style'] });
    render();
  }

  function flash(el, text, label) {
    navigator.clipboard.writeText(text);
    const old = el.innerHTML;
    el.innerHTML = ICON.check + (label ? `<span>${label}</span>` : '');
    setTimeout(() => (el.innerHTML = old), 1200);
  }

  function applyTheme() { if (root) root.querySelector('.wrap').classList.toggle('dark', isDark()); }

  function esc(s) { return String(s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c])); }

  function render() {
    if (!root) return;
    applyTheme();
    const $ = s => root.querySelector(s);
    $('.fab .n').textContent = queries.size || sources.size ? `${queries.size} · ${sources.size}` : '';
    const [qb, sb] = root.querySelectorAll('.seg button');
    qb.classList.toggle('on', tab === 'queries'); sb.classList.toggle('on', tab === 'sources');
    qb.querySelector('.c').textContent = queries.size || '';
    sb.querySelector('.c').textContent = sources.size || '';
    const list = $('.list');
    if (tab === 'queries') {
      list.innerHTML = queries.size
        ? [...queries].map((q, i) => `
          <a class="row" href="https://www.google.com/search?q=${encodeURIComponent(q)}" target="_blank" rel="noopener" title="Run on Google">
            <span class="idx">${i + 1}</span>
            <span class="main"><div class="t">${esc(q)}</div></span>
            <button class="act" data-copy="${esc(q)}" title="Copy">${ICON.copy}</button>
          </a>`).join('')
        : `<div class="empty"><b>No queries captured</b>ChatGPT hides queries in saved chats.<br>Send a new message with search on.</div>`;
    } else {
      list.innerHTML = sources.size
        ? [...sources].map(([u, s]) => `
          <a class="row" href="${esc(u)}" target="_blank" rel="noopener">
            <img src="https://www.google.com/s2/favicons?domain=${encodeURIComponent(s.domain)}&sz=64" alt="">
            <span class="main"><div class="t">${esc(s.title)}</div><div class="s">${esc(s.domain)}</div></span>
            <span class="act">${ICON.ext}</span>
          </a>`).join('')
        : `<div class="empty"><b>No sources yet</b>Sources show up when ChatGPT searches the web.</div>`;
    }
  }

  const iv = setInterval(() => { if (document.body) { build(); clearInterval(iv); } }, 300);
})();
